// Нысандарды іздеу және сүзгілеу функциясы
function filterLocations() {
    let input = document.getElementById('searchInput').value.toLowerCase();
    let items = document.querySelectorAll('.loc-item');

    items.forEach(item => {
        let title = item.querySelector('h4').textContent.toLowerCase();
        let address = item.querySelector('p').textContent.toLowerCase();

        if (title.includes(input) || address.includes(input)) {
            item.style.display = "";
        } else {
            item.style.display = "none";
        }
    });
}
